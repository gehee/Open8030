cmake_minimum_required(VERSION 3.20.0)
list(APPEND BOARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
list(APPEND SOC_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
list(APPEND DTS_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
if(BOARD MATCHES "^ar8030_bringup")
  list(APPEND EXTRA_CONF_FILE ${CMAKE_CURRENT_SOURCE_DIR}/boards/ar8030_bringup.conf)
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(ar8030_zephyr)
target_sources(app PRIVATE src/main.c src/frame.c src/dispatch.c)
if(CONFIG_SOC_AR8030_BRINGUP)
  target_sources(app PRIVATE src/sdio_device.c src/rf_device.c src/mac.c src/dev.c src/binding.c src/dev_tx.c src/dev_mcs.c src/fch_monitor.c src/channel_monitor.c src/socket_rx.c src/event.c src/boot.c src/json.c src/config.c src/power_config.c)
  target_sources(app PRIVATE src/power.c src/power_control.c src/flash.c src/flash_spi.c src/sfdp.c src/factory.c src/factory_device.c src/ap_time.c src/command.c src/scan.c src/scan_scores.c src/tune.c src/phy_control.c src/phy_receive.c)
  target_sources(app PRIVATE src/tx_mcs.c src/interleaver.c src/tx_irq.c)
  target_sources(app PRIVATE src/telemetry.c src/data_framing.c src/data_event.c)
  target_compile_definitions(app PRIVATE AR_SDIO_STOCK_MEMORY_ATTRIBUTES=1)
  set(AR8030_BOOT_PROFILE_SOURCE "" CACHE FILEPATH "Explicit private generated board profile source")
  if(AR8030_BOOT_PROFILE_SOURCE)
    if(NOT EXISTS "${AR8030_BOOT_PROFILE_SOURCE}")
      message(FATAL_ERROR "Requested private boot profile source does not exist")
    endif()
    target_sources(app PRIVATE "${AR8030_BOOT_PROFILE_SOURCE}")
    target_compile_definitions(app PRIVATE AR_BOOT_EMBEDDED=1)
  endif()
endif()
target_include_directories(app PRIVATE include)
